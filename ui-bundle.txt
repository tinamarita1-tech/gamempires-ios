/** Username + password, sent once to the server to get an app password (the password itself is never stored). */
@Composable
private fun PasswordLoginForm(
    onPasswordLogin: (String, String) -> Unit,
    onBrowserLogin: () -> Unit
) {
    var username by androidx.compose.runtime.saveable.rememberSaveable { androidx.compose.runtime.mutableStateOf("") }
    // Not saveable on purpose: the password is never written to saved state.
    var password by androidx.compose.runtime.remember { androidx.compose.runtime.mutableStateOf("") }
    var showPassword by androidx.compose.runtime.remember { androidx.compose.runtime.mutableStateOf(false) }
    val canSubmit = username.isNotBlank() && password.isNotEmpty()
    val submit = { if (canSubmit) onPasswordLogin(username.trim(), password) }
    val fieldColors = androidx.compose.material3.OutlinedTextFieldDefaults.colors(
        focusedBorderColor = GamEmpiresGold,
        unfocusedBorderColor = GamEmpiresCard,
        focusedTextColor = GamEmpiresTextPrimary,
        unfocusedTextColor = GamEmpiresTextPrimary,
        cursorColor = GamEmpiresGold,
        focusedLabelColor = GamEmpiresGold,
        unfocusedLabelColor = GamEmpiresTextSecondary
    )

    androidx.compose.material3.OutlinedTextField(
        value = username,
        onValueChange = { username = it },
        label = { Text("Username") },
        singleLine = true,
        colors = fieldColors,
        shape = RoundedCornerShape(12.dp),
        keyboardOptions = androidx.compose.foundation.text.KeyboardOptions(
            keyboardType = androidx.compose.ui.text.input.KeyboardType.Ascii,
            imeAction = androidx.compose.ui.text.input.ImeAction.Next,
            autoCorrectEnabled = false
        ),
        modifier = Modifier.fillMaxWidth()
    )
    Spacer(modifier = Modifier.height(12.dp))
    androidx.compose.material3.OutlinedTextField(
        value = password,
        onValueChange = { password = it },
        label = { Text("Password") },
        singleLine = true,
        colors = fieldColors,
        shape = RoundedCornerShape(12.dp),
        visualTransformation = if (showPassword) androidx.compose.ui.text.input.VisualTransformation.None
            else androidx.compose.ui.text.input.PasswordVisualTransformation(),
        keyboardOptions = androidx.compose.foundation.text.KeyboardOptions(
            keyboardType = androidx.compose.ui.text.input.KeyboardType.Password,
            imeAction = androidx.compose.ui.text.input.ImeAction.Done
        ),
        keyboardActions = androidx.compose.foundation.text.KeyboardActions(onDone = { submit() }),
        trailingIcon = {
            androidx.compose.material3.IconButton(onClick = { showPassword = !showPassword }) {
                Icon(
                    imageVector = if (showPassword) androidx.compose.material.icons.Icons.Default.VisibilityOff
                        else androidx.compose.material.icons.Icons.Default.Visibility,
                    contentDescription = if (showPassword) "Hide password" else "Show password",
                    tint = GamEmpiresTextSecondary
                )
            }
        },
        modifier = Modifier.fillMaxWidth()
    )
    Spacer(modifier = Modifier.height(20.dp))
    Button(
        onClick = { submit() },

import androidx.lifecycle.ViewModel

sealed interface LoginUiState {
    data object Idle : LoginUiState
    data object Initializing : LoginUiState
    data object SigningIn : LoginUiState
    data class AwaitingBrowserAuth(val loginUrl: String) : LoginUiState
    data object Success : LoginUiState
    data class Error(val message: String) : LoginUiState
}

@HiltViewModel
class LoginViewModel @Inject constructor(
    private val api: NextcloudTalkApi,
    private val authStorage: NextcloudAuthStorage,
    private val talkRepository: NextcloudTalkRepository,
    private val pushRegistrar: com.gamempires.messenger.push.PushRegistrar,
    private val sessionGuard: com.gamempires.messenger.push.SessionGuard,
    baseClient: okhttp3.OkHttpClient
) : ViewModel() {

    // Plain client (no stored-login interceptor) for exchanging the typed password once.
    private val loginClient: okhttp3.OkHttpClient = baseClient.newBuilder().apply {
        interceptors().removeAll { it is com.gamempires.messenger.data.nextcloud.interceptor.NextcloudAuthInterceptor }
    }.build()

    val isLoggedIn: StateFlow<Boolean> = authStorage.isLoggedIn

    private val _uiState = MutableStateFlow<LoginUiState>(LoginUiState.Idle)
    val uiState: StateFlow<LoginUiState> = _uiState.asStateFlow()

    private var pollJob: Job? = null

    init {
        viewModelScope.launch {
            talkRepository.isSessionRevoked.collect { revoked ->
                if (revoked) {
                    _uiState.value = LoginUiState.Error("Your login session was revoked on the server. Please log in again.")
                }
            }
        }
        viewModelScope.launch {
            sessionGuard.signedOutElsewhere.collect { elsewhere ->
                if (elsewhere) {
                    _uiState.value = LoginUiState.Error(
                        "You were signed out because your account was signed in on another phone. " +
                            "Only one phone can use your account at a time."
                    )
                }
            }
        }
    }

    fun startLoginFlow(serverUrl: String = NextcloudAuthStorage.DEFAULT_SERVER_URL) {
        pollJob?.cancel()
        sessionGuard.clearSignedOutNotice()
        _uiState.value = LoginUiState.Initializing

        viewModelScope.launch {
            try {
                val initResponse = api.initLoginV2()
                val loginUrl = initResponse.login
                val pollUrl = initResponse.poll.endpoint
                val token = initResponse.poll.token

                _uiState.value = LoginUiState.AwaitingBrowserAuth(loginUrl)

                // Start polling Nextcloud Login Flow v2
                pollJob = launch {
                    while (isActive) {
                        delay(2000)
                        try {
                            val pollResp = api.pollLoginV2(pollUrl, token)
                            if (pollResp.isSuccessful && pollResp.body() != null) {
                                val successData = pollResp.body()!!
                                authStorage.saveCredentials(
                                    serverUrl = successData.server.ifBlank { serverUrl },
                                    loginName = successData.loginName,
                                    appPassword = successData.appPassword
                                )
                                // This phone becomes the account's only mobile session; the previous phone is signed out.
                                sessionGuard.claim()
                                talkRepository.refreshRooms()
                                _uiState.value = LoginUiState.Success
                                break
                            }
                        } catch (_: Exception) {
                            // 404 or connection pending, continue polling
                        }
                    }
                }
            } catch (e: Exception) {
                _uiState.value = LoginUiState.Error(e.message ?: "Failed to connect to Nextcloud")
            }
        }
    }

    /**
     * In-app login: trades the username + password for a Nextcloud app password once
     * (GET /ocs/v2.php/core/getapppassword). Only the app password is stored; the typed password
     * is never saved. Accounts with two-step login must use the browser login instead.
     */
    fun loginWithPassword(username: String, password: String) {
        val user = username.trim()
        if (user.isEmpty() || password.isEmpty()) return
        pollJob?.cancel()
        sessionGuard.clearSignedOutNotice()
        _uiState.value = LoginUiState.SigningIn
        viewModelScope.launch {
            val server = NextcloudAuthStorage.DEFAULT_SERVER_URL
            val result = runCatching {
                kotlinx.coroutines.withContext(kotlinx.coroutines.Dispatchers.IO) {
                    val request = okhttp3.Request.Builder()
                        .url("$server/ocs/v2.php/core/getapppassword?format=json")
                        .header("Authorization", okhttp3.Credentials.basic(user, password, Charsets.UTF_8))
                        .header("OCS-APIRequest", "true")
                        .header("Accept", "application/json")
                        // Names the session in Nextcloud's "Devices & sessions" list.
                        .header("User-Agent", "GamEmpires Messenger (Android ${android.os.Build.MODEL})")
                        .build()
                    loginClient.newCall(request).execute().use { response ->
                        val body = response.body.string()
                        val appPassword = runCatching {
                            org.json.JSONObject(body).getJSONObject("ocs").getJSONObject("data").getString("apppassword")
                        }.getOrNull()
                        response.code to appPassword
                    }
                }
            }
            val (code, appPassword) = result.getOrElse {
                _uiState.value = LoginUiState.Error("Can't reach the chat server. Check your internet connection and try again.")
                return@launch
            }
            if (code == 200 && !appPassword.isNullOrBlank()) {
                authStorage.saveCredentials(serverUrl = server, loginName = user, appPassword = appPassword)
                sessionGuard.claim()
                talkRepository.refreshRooms()
                _uiState.value = LoginUiState.Success
            } else {
                _uiState.value = LoginUiState.Error(
                    when (code) {
                        401, 403 -> "Wrong username or password. If two-step login is turned on for your account, use \"Log in with browser\" below."
                        429 -> "Too many attempts. Please wait a minute and try again."
                        else -> "Login failed (server answered $code). Please try again."
                    }
                )
            }
        }
    }

    fun logout() {
        pollJob?.cancel()
        viewModelScope.launch {
            // Stop notifications for this phone before the app password is deleted.
            pushRegistrar.unregister()
            talkRepository.logout()
            _uiState.value = LoginUiState.Idle
        }
    }

    fun dismiss() {
        pollJob?.cancel()
        _uiState.value = LoginUiState.Idle
    }
}

@HiltViewModel
class ChatListViewModel @Inject constructor(
    private val chatRepository: ChatRepository
) : ViewModel() {

    private val _selectedFilterTab = MutableStateFlow(ChatFilterTab.HOME)
    private val _selectedChatId = MutableStateFlow<String?>(null)
    private val _isManualRefreshing = MutableStateFlow(false)
    private val _isBackgroundSyncing = MutableStateFlow(false)

    /** Home tab status when the Announcements room's last item is a system note (e.g. "picture changed"). */
    private val _announcementStatus = MutableStateFlow<Pair<String, com.gamempires.messenger.model.Message>?>(null)
    private var announcementCheckedKey: String? = null

    init {
        viewModelScope.launch {
            chatRepository.chats.collect { chats ->
                val room = chats.firstOrNull {
                    it.type == ChatType.GROUP && it.title.trim().equals("Announcements", ignoreCase = true)
                } ?: return@collect
                val last = room.lastMessage
                if (last != null && !last.isSystemMessage && !last.isDeleted) {
                    _announcementStatus.value = room.id to last
                    announcementCheckedKey = room.id + ":" + last.id
                    return@collect
                }
                val key = room.id + ":" + (last?.id ?: "none")
                if (key == announcementCheckedKey) return@collect   // already looked up for this state
                announcementCheckedKey = key
                val real = chatRepository.latestUserMessage(room.id)
                _announcementStatus.value = real?.let { room.id to it }
            }
        }
    }

    private val group1Flow = combine(
        chatRepository.filteredChats,
        chatRepository.chats,
        chatRepository.searchQuery,
        _selectedFilterTab
    ) { filtered, allChats, query, filterTab ->
        FlowGroup1(filtered, allChats, query, filterTab)
    }

    private val group2Flow = combine(
        chatRepository.currentUser,
        chatRepository.userPresences,
        chatRepository.typingUsers,
        chatRepository.users,
        _selectedChatId
    ) { currentUsr, presences, typingMap, allUsers, selectedId ->
        FlowGroup2(currentUsr, presences, typingMap, allUsers, selectedId)
    }

    private val statusFlow = combine(
        chatRepository.isLoggedIn,
        _isManualRefreshing,
        _isBackgroundSyncing,
        chatRepository.isOffline,
        chatRepository.isSessionRevoked
    ) { loggedIn, manualRefreshing, backgroundSyncing, offline, revoked ->
        StatusGroup(loggedIn, manualRefreshing, backgroundSyncing, offline, revoked)
    }

    val uiState: StateFlow<ChatListUiState> = combine(
        group1Flow,
        group2Flow,
        statusFlow,
        _announcementStatus
    ) { g1, g2, status, announcement ->
        val counts = mapOf(
            ChatFilterTab.HOME to 0,
            ChatFilterTab.ALL to g1.filteredChats.size,
            ChatFilterTab.DIRECT to g1.filteredChats.count { it.type == ChatType.DIRECT },
            ChatFilterTab.GROUPS to g1.filteredChats.count { it.type == ChatType.GROUP },
            ChatFilterTab.UNREAD to g1.filteredChats.count { it.unreadCount > 0 }
        )

        val byTab = ChatFilterTab.entries.associateWith { tab ->
            when (tab) {
                ChatFilterTab.HOME, ChatFilterTab.ALL -> g1.filteredChats
                ChatFilterTab.DIRECT -> g1.filteredChats.filter { it.type == ChatType.DIRECT }
                ChatFilterTab.GROUPS -> g1.filteredChats.filter { it.type == ChatType.GROUP }
                ChatFilterTab.UNREAD -> g1.filteredChats.filter { it.unreadCount > 0 }
            }
        }
        val displayChats = byTab[g1.filterTab] ?: g1.filteredChats

        val unreadTotal = g1.allChats.sumOf { it.unreadCount }

        ChatListUiState(
            chats = displayChats,
            chatsByTab = byTab,
            searchQuery = g1.searchQuery,
            selectedFilterTab = g1.filterTab,
            currentUser = g2.currentUser,
            userPresences = g2.userPresences,
            typingUsers = g2.typingUsers,
            users = g2.users,
            selectedChatId = g2.selectedChatId,
            filterCounts = counts,
            statusMessage = announcement,
            totalUnreadCount = unreadTotal,
            isLoggedIn = status.isLoggedIn,
            isRefreshing = status.isManualRefreshing,
            isManualRefreshing = status.isManualRefreshing,
            isBackgroundSyncing = status.isBackgroundSyncing,
            isOffline = status.isOffline,
            isSessionRevoked = status.isSessionRevoked
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = ChatListUiState()
    )

    fun manualRefresh() {
        viewModelScope.launch {
            _isManualRefreshing.value = true
            try {
                chatRepository.refreshRooms()
            } finally {
                _isManualRefreshing.value = false
            }
        }
    }

    fun backgroundSync() {
        viewModelScope.launch {
            _isBackgroundSyncing.value = true
            try {
                chatRepository.refreshRooms()
            } finally {
                _isBackgroundSyncing.value = false
            }
        }
    }

    fun refresh() {

    }

    suspend fun searchUsers(query: String): List<User> {

private fun getMessagePreview(content: String, attachmentTypes: List<AttachmentType>): String {
    if (attachmentTypes.isNotEmpty()) {
        val icon = when (attachmentTypes.first()) {
            AttachmentType.IMAGE -> "📷 Photo"
            AttachmentType.VIDEO -> "🎥 Video"
            AttachmentType.AUDIO -> "🎵 Audio"
            AttachmentType.DOCUMENT -> "📄 Document"
            AttachmentType.FILE -> "📁 File"
        }
        return if (content.isNotBlank()) "$icon: $content" else icon
    }
    return content
}

private fun formatMessageTime(timestamp: Long): String {
    val now = Calendar.getInstance()
    val msgTime = Calendar.getInstance().apply { timeInMillis = timestamp }

    return if (now.get(Calendar.YEAR) == msgTime.get(Calendar.YEAR) &&
        now.get(Calendar.DAY_OF_YEAR) == msgTime.get(Calendar.DAY_OF_YEAR)
    ) {
        SimpleDateFormat("h:mm a", Locale.getDefault()).format(Date(timestamp))
    } else if (now.get(Calendar.YEAR) == msgTime.get(Calendar.YEAR) &&
        now.get(Calendar.DAY_OF_YEAR) - msgTime.get(Calendar.DAY_OF_YEAR) == 1
    ) {
        "Yesterday"
    } else if (now.timeInMillis - timestamp < 7 * 24 * 60 * 60 * 1000L) {
        SimpleDateFormat("EEE", Locale.getDefault()).format(Date(timestamp))
    } else {
        SimpleDateFormat("MMM d", Locale.getDefault()).format(Date(timestamp))
    }
}

@Composable
fun UserAvatar(
    imageUrl: String?,
    name: String,
    presenceStatus: PresenceStatus? = null,
    chatType: ChatType = ChatType.DIRECT,
    size: Dp = 48.dp,
    showPresence: Boolean = true,
    modifier: Modifier = Modifier
) {
    val badgeSize = (size.value * 0.28f).dp.coerceAtLeast(10.dp)
    val badgeOffset = (size.value * 0.02f).dp

    Box(

            SubcomposeAsyncImage(
                model = imageUrl,
                contentDescription = name,
                contentScale = ContentScale.Crop,
                modifier = Modifier
                    .size(size)
                    .clip(AvatarTileShape),
                loading = {
                    AvatarFallback(name = name, chatType = chatType, size = size)

                error = {
                    AvatarFallback(name = name, chatType = chatType, size = size)
                }
            )
        } else {
            AvatarFallback(name = name, chatType = chatType, size = size)
        }

        if (showPresence && presenceStatus != null && chatType == ChatType.DIRECT) {
            PresenceBadge(
                status = presenceStatus,
                size = badgeSize,
                modifier = Modifier
                    .align(Alignment.BottomEnd)
                    .offset(x = badgeOffset, y = badgeOffset)
            )
        }
    }

/**
 * Background container rendering the repeating subtle "GE" watermark pattern
 * seen across the GamEmpires desktop/adaptive view in Image 2.
 */
@Composable
fun GamEmpiresWatermarkBackground(
    modifier: Modifier = Modifier,
    backgroundColor: Color = GamEmpiresBg,
    watermarkColor: Color = Color(0x0CFFFFFF),
    spacing: Dp = 140.dp,
    content: @Composable () -> Unit
) {
    val textMeasurer = rememberTextMeasurer()
    val watermarkStyle = TextStyle(
        fontSize = 44.sp,
        fontWeight = FontWeight.Black,
        color = watermarkColor,
        fontFamily = FontFamily.SansSerif
    )

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(backgroundColor)
    ) {
        Canvas(modifier = Modifier.fillMaxSize()) {
            val stepPx = spacing.toPx()
            val textLayout = textMeasurer.measure(
                text = "GE",
                style = watermarkStyle
            )

            var y = -stepPx
            while (y < size.height + stepPx) {
                var x = -stepPx
                while (x < size.width + stepPx) {
                    rotate(degrees = -15f, pivot = Offset(x, y)) {
                        drawText(
                            textLayoutResult = textLayout,
                            topLeft = Offset(x, y)
                        )
                    }
                    x += stepPx
                }
                y += stepPx * 0.9f
            }
        }

        content()
    }
}

@Composable
fun MarkdownText(
    markdown: String,
    modifier: Modifier = Modifier,
    color: Color = MaterialTheme.colorScheme.onSurface,
    linkColor: Color = MaterialTheme.colorScheme.primary,
    fontSize: TextUnit = 14.sp,
    lineHeight: TextUnit = 20.sp
) {
    val annotatedString = remember(markdown, color, linkColor, fontSize) {
        buildMarkdownAnnotatedString(markdown, baseFontSize = fontSize, linkColor = linkColor, codeColor = color)
    }

    Text(
        text = annotatedString,
        modifier = modifier,
        style = MaterialTheme.typography.bodyMedium.copy(
            color = color,
            fontSize = fontSize,
            lineHeight = lineHeight
        )
    )
}

fun buildMarkdownAnnotatedString(
    markdown: String,
    baseFontSize: TextUnit = 14.sp,
    linkColor: Color = Color(0xFF60A5FA),
    codeColor: Color = Color(0xFFFDE68A)
): AnnotatedString {
    return buildAnnotatedString {
        val lines = markdown.split("\n")
        lines.forEachIndexed { index, line ->
            var lineText = line
            var isHeading1 = false
            var isHeading2 = false
            var isHeading3 = false
            var isListItem = false

            if (lineText.startsWith("# ")) {
                isHeading1 = true
                lineText = lineText.removePrefix("# ")
            } else if (lineText.startsWith("## ")) {
                isHeading2 = true
                lineText = lineText.removePrefix("## ")
            } else if (lineText.startsWith("### ")) {
                isHeading3 = true
                lineText = lineText.removePrefix("### ")
            } else if (lineText.startsWith("* ") || lineText.startsWith("- ")) {
                isListItem = true
                lineText = "• " + lineText.substring(2)
            } else if (lineText.matches(Regex("^\\d+\\.\\s+.*"))) {
                isListItem = true
            }

            val lineStart = length
            append(lineText)
            val lineEnd = length

            // Apply line-level styles (headings, list items)
            when {
                isHeading1 -> {
                    addStyle(
                        SpanStyle(
                            fontSize = (baseFontSize.value + 4).sp,
                            fontWeight = FontWeight.Bold,
                            color = GamEmpiresGold
                        ),
                        lineStart,
                        lineEnd
                    )
                }
                isHeading2 -> {
                    addStyle(
                        SpanStyle(
                            fontSize = (baseFontSize.value + 2).sp,
                            fontWeight = FontWeight.Bold,
                            color = GamEmpiresGold
                        ),
                        lineStart,
                        lineEnd
                    )
                }
                isHeading3 -> {
                    addStyle(
                        SpanStyle(
                            fontSize = (baseFontSize.value + 1).sp,
                            fontWeight = FontWeight.SemiBold,
                            color = GamEmpiresGold
                        ),
                        lineStart,
                        lineEnd
                    )
                }
                isListItem -> {
                    addStyle(
                        SpanStyle(fontWeight = FontWeight.Normal),
                        lineStart,
                        lineEnd
                    )
                }
            }

            val linkStyles = TextLinkStyles(
                style = SpanStyle(
                    color = linkColor,
                    textDecoration = TextDecoration.Underline,
                    fontWeight = FontWeight.Medium
                )
            )

            // 1) Markdown Links: [text](url)
            val linkRegex = Regex("\\[([^]]+)]\\(([^)]+)\\)")
            for (match in linkRegex.findAll(lineText)) {
                val url = match.groupValues[2]
                val matchStart = lineStart + match.range.first
                val matchEnd = lineStart + match.range.last + 1
                try {
                    addLink(LinkAnnotation.Url(url = url, styles = linkStyles), matchStart, matchEnd)
                } catch (_: Exception) {}
            }

            // 2) Bare URLs: https?://\S+
            val bareUrlRegex = Regex("(?<!\\()https?://[a-zA-Z0-9./_?=&%#+-]+")
            for (match in bareUrlRegex.findAll(lineText)) {
                val url = match.value
                val matchStart = lineStart + match.range.first
                val matchEnd = lineStart + match.range.last + 1
                try {
                    addLink(LinkAnnotation.Url(url = url, styles = linkStyles), matchStart, matchEnd)
                } catch (_: Exception) {}
            }

            // 3) @mentions: @[a-zA-Z0-9_.-]+
            val mentionRegex = Regex("@[a-zA-Z0-9_.-]+")
            for (match in mentionRegex.findAll(lineText)) {
                val matchStart = lineStart + match.range.first
                val matchEnd = lineStart + match.range.last + 1
                addStyle(
                    SpanStyle(
                        color = linkColor,
                        fontWeight = FontWeight.Bold
                    ),
                    matchStart,
                    matchEnd
                )
            }

            // 4) Bold: \*\*([^*]+)\*\*
            val boldRegex = Regex("\\*\\*([^*]+)\\*\\*")
            for (match in boldRegex.findAll(lineText)) {
                val matchStart = lineStart + match.range.first
                val matchEnd = lineStart + match.range.last + 1
                addStyle(
                    SpanStyle(fontWeight = FontWeight.Bold),
                    matchStart,
                    matchEnd
                )
            }

            // 5) Italic: \*([^*]+)\*
            val italicRegex = Regex("(?<!\\*)\\*([^*]+)\\*(?!\\*)")
            for (match in italicRegex.findAll(lineText)) {
                val matchStart = lineStart + match.range.first
                val matchEnd = lineStart + match.range.last + 1
                addStyle(
                    SpanStyle(fontStyle = FontStyle.Italic),
                    matchStart,
                    matchEnd
                )
            }

            // 6) Inline code: `([^`]+)`
            val codeRegex = Regex("`([^`]+)`")
            for (match in codeRegex.findAll(lineText)) {
                val matchStart = lineStart + match.range.first
                val matchEnd = lineStart + match.range.last + 1
                addStyle(
                    SpanStyle(
                        fontFamily = FontFamily.Monospace,
                        background = Color(0x33000000),
                        color = codeColor
                    ),
                    matchStart,
                    matchEnd
                )
            }

            if (index < lines.size - 1) {
                append("\n")
            }
        }
    }
}

                                text = if (isSearching && inChatSearchQuery.isNotBlank()) "No messages match your search" else "No messages yet. Send a greeting!",
                                color = GamEmpiresTextMuted,
                                fontSize = 14.sp
                            )
                        } else {
                            CircularProgressIndicator(
                                color = MaterialTheme.colorScheme.primary,
                                strokeWidth = 2.dp,
                                modifier = Modifier.size(28.dp)
                            )
                        }
                    }
                } else {
                    LazyColumn(
                        state = listState,
                        reverseLayout = true,
                        contentPadding = PaddingValues(horizontal = 16.dp, vertical = 12.dp),
                        verticalArrangement = Arrangement.spacedBy(8.dp, Alignment.Bottom),
                        modifier = Modifier.fillMaxSize()
                    ) {
                        items(
                            items = newestFirst,
                            key = { it.id },
                            contentType = { if (it.isSystemMessage) 0 else 1 }
                        ) { message ->
                            // Requirement 4: System messages show as a small centered grey line
                            if (message.isSystemMessage) {
                                SystemMessageLine(content = message.content)
                            } else {
                                val isMe = message.senderId == uiState.currentUser?.id
                                val quotedMessage = if (message.replyToMessageId != null) {
                                    messagesById[message.replyToMessageId]
                                } else null

                                MessageBubble(
                                    message = message,
                                    quotedMessage = quotedMessage,
                                    isMe = isMe,
                                    downloadProgress = uiState.downloadProgress,
                                    downloadedAttachmentIds = uiState.downloadedAttachmentIds,
                                    onLongClick = { selectedMessageForMenu = message },
                                    onAddReaction = { emoji -> onAddReaction(message.id, emoji) },
                                    onImageClick = { att, url -> fullScreenAttachment = att to url },
                                    onDownloadClick = onDownloadAttachment,
                                    onSaveToDownloads = onSaveToDownloads
                                )
                            }
                        }
                    }
                }
            }

            // Divider
            Box(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(1.dp)
                    .background(GamEmpiresBorder)
            )

            // Bottom Input Bar with Reply Quote Banner
            androidx.compose.runtime.key(uiState.chat?.id) {
            ChatInputBar(
                text = uiState.inputText,
                replyingToMessage = uiState.replyingToMessage,
                uploadProgress = uiState.uploadProgress,
                pendingFiles = pendingFiles,
                onRemovePendingFile = { pendingFiles.remove(it) },
                onCancelReply = { onSetReplyingTo(null) },
                onTextChange = onInputTextChange,
                onAttachClick = { showAttachmentOptions = true },
                onSendMessage = sendAll
            )
            }
        }
    }

    // Attachment Chooser Bottom Sheet (Photos/Videos or Documents/Files)
    if (showAttachmentOptions) {
        ModalBottomSheet(
            onDismissRequest = { showAttachmentOptions = false },
            containerColor = GamEmpiresSurface
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 24.dp, vertical = 16.dp)
            ) {
                Text(
                    text = "Share Attachment",
                    fontSize = 18.sp,
                    fontWeight = FontWeight.Bold,
                    color = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                    modifier = Modifier.padding(bottom = 16.dp)
                )

                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(RoundedCornerShape(12.dp))
                        .clickable {
                            showAttachmentOptions = false
                            photoPickerLauncher.launch(
                                PickVisualMediaRequest(ActivityResultContracts.PickVisualMedia.ImageAndVideo)
                            )
                        }
                        .padding(vertical = 12.dp, horizontal = 8.dp)
                ) {
                    Icon(
                        imageVector = Icons.Default.PhotoLibrary,
                        contentDescription = "Photos & Videos",
                        tint = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                        modifier = Modifier.size(24.dp)
                    )
                    Spacer(modifier = Modifier.width(16.dp))
                    Column {
                        Text(
                            text = "Photos & Videos",
                            fontSize = 15.sp,
                            fontWeight = FontWeight.SemiBold,
                            color = MaterialTheme.colorScheme.onSurface
                        )
                        Text(
                            text = "Pick one or more images or videos",
                            fontSize = 12.sp,
                            color = GamEmpiresTextSecondary
                        )
                    }
                }

                Spacer(modifier = Modifier.height(8.dp))

                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .clip(RoundedCornerShape(12.dp))
                        .clickable {
                            showAttachmentOptions = false
                            filePickerLauncher.launch("*/*")
                        }
                        .padding(vertical = 12.dp, horizontal = 8.dp)
                ) {
                    Icon(
                        imageVector = Icons.Default.Description,
                        contentDescription = "Document or File",
                        tint = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                        modifier = Modifier.size(24.dp)
                    )
                    Spacer(modifier = Modifier.width(16.dp))
                    Column {
                        Text(
                            text = "Document / File",
                            fontSize = 15.sp,
                            fontWeight = FontWeight.SemiBold,
                            color = MaterialTheme.colorScheme.onSurface
                        )
                        Text(
                            text = "Share PDF, APK, documents, zip, or any file",
                            fontSize = 12.sp,
                            color = GamEmpiresTextSecondary
                        )
                    }
                }

                Spacer(modifier = Modifier.height(24.dp))
            }
        }
    }

    // Long-Press Bottom Sheet Context Menu
    if (selectedMessageForMenu != null) {
        val targetMsg = selectedMessageForMenu!!
        val isOwnMessage = targetMsg.senderId == uiState.currentUser?.id

        ModalBottomSheet(
            onDismissRequest = { selectedMessageForMenu = null },
            containerColor = GamEmpiresSurface
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 24.dp, vertical = 16.dp)
            ) {
                Text(
                    text = "React with emoji",
                    style = MaterialTheme.typography.labelMedium,
                    color = GamEmpiresTextSecondary,
                    fontWeight = FontWeight.SemiBold
                )

                Spacer(modifier = Modifier.height(12.dp))

                // Quick emoji reaction bar
                Row(
                    horizontalArrangement = Arrangement.SpaceBetween,
                    modifier = Modifier.fillMaxWidth()
                ) {
                    val quickEmojis = listOf("👍", "❤️", "🔥", "😂", "🎉", "👏")
                    quickEmojis.forEach { emoji ->
                        Surface(
                            shape = CircleShape,
                            color = GamEmpiresCard,
                            modifier = Modifier
                                .size(48.dp)
                                .clip(CircleShape)
                                .clickable {
                                    onAddReaction(targetMsg.id, emoji)
                                    selectedMessageForMenu = null
                                }
                        ) {
                            Box(contentAlignment = Alignment.Center) {
                                Text(text = emoji, fontSize = 22.sp)
                            }
                        }
                    }
                }

                Spacer(modifier = Modifier.height(20.dp))

                // Action: Reply
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                        .clickable {
                            onSetReplyingTo(targetMsg)
                            selectedMessageForMenu = null
                        }
                ) {
                    Icon(
                        imageVector = Icons.AutoMirrored.Filled.Reply,
                        contentDescription = "Reply",
                        tint = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                        modifier = Modifier.size(22.dp)
                    )
                    Spacer(modifier = Modifier.width(16.dp))
                    Text(
                        text = "Reply",
                        color = GamEmpiresTextPrimary,
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium
                    )
                }

                // Action: Forward
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                        .clickable {
                            messageToForward = targetMsg
                            selectedMessageForMenu = null
                        }
                ) {
                    Icon(
                        imageVector = Icons.AutoMirrored.Filled.Forward,
                        contentDescription = "Forward",
                        tint = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                        modifier = Modifier.size(22.dp)
                    )
                    Spacer(modifier = Modifier.width(16.dp))
                    Text(
                        text = "Forward",
                        color = GamEmpiresTextPrimary,
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium
                    )
                }

                // Action: Save to Downloads (if message has attachments)
                if (targetMsg.attachments.isNotEmpty()) {
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(48.dp)
                            .clickable {
                                targetMsg.attachments.forEach { att ->
                                    onSaveToDownloads(att)
                                }
                                selectedMessageForMenu = null
                            }
                    ) {
                        Icon(
                            imageVector = Icons.Default.Download,
                            contentDescription = "Save to Downloads",
                            tint = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                            modifier = Modifier.size(22.dp)
                        )
                        Spacer(modifier = Modifier.width(16.dp))
                        Text(
                            text = "Save to Downloads",
                            color = GamEmpiresTextPrimary,
                            fontSize = 15.sp,
                            fontWeight = FontWeight.Medium
                        )
                    }
                }

                // Action: Share via...
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                        .clickable {
                            onShareMessage(targetMsg)
                            selectedMessageForMenu = null
                        }
                ) {
                    Icon(
                        imageVector = Icons.Default.Share,
                        contentDescription = "Share via...",
                        tint = GamEmpiresTextSecondary,
                        modifier = Modifier.size(22.dp)
                    )
                    Spacer(modifier = Modifier.width(16.dp))
                    Text(
                        text = "Share via...",
                        color = GamEmpiresTextPrimary,
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium
                    )
                }

                // Action: Copy Text
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(48.dp)
                        .clickable {
                            val clipboard = context.getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
                            val clip = ClipData.newPlainText("Message Text", targetMsg.content)
                            clipboard.setPrimaryClip(clip)
                            Toast.makeText(context, "Text copied to clipboard", Toast.LENGTH_SHORT).show()
                            selectedMessageForMenu = null
                        }
                ) {
                    Icon(
                        imageVector = Icons.Default.ContentCopy,
                        contentDescription = "Copy text",
                        tint = GamEmpiresTextSecondary,
                        modifier = Modifier.size(22.dp)
                    )
                    Spacer(modifier = Modifier.width(16.dp))
                    Text(
                        text = "Copy text",
                        color = GamEmpiresTextPrimary,
                        fontSize = 15.sp,
                        fontWeight = FontWeight.Medium
                    )
                }

                // Action: Edit (own messages only)
                if (isOwnMessage) {
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(48.dp)
                            .clickable {
                                messageToEdit = targetMsg
                                selectedMessageForMenu = null
                            }
                    ) {
                        Icon(
                            imageVector = Icons.Default.Edit,
                            contentDescription = "Edit message",
                            tint = GamEmpiresTextSecondary,
                            modifier = Modifier.size(22.dp)
                        )
                        Spacer(modifier = Modifier.width(16.dp))
                        Text(
                            text = "Edit message",
                            color = GamEmpiresTextPrimary,
                            fontSize = 15.sp,
                            fontWeight = FontWeight.Medium
                        )
                    }

                    // Action: Delete (own messages only)
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier
                            .fillMaxWidth()
                            .height(48.dp)
                            .clickable {
                                messageToDelete = targetMsg
                                selectedMessageForMenu = null
                            }
                    ) {
                        Icon(
                            imageVector = Icons.Default.Delete,
                            contentDescription = "Delete message",
                            tint = Color(0xFFEF4444),
                            modifier = Modifier.size(22.dp)
                        )
                        Spacer(modifier = Modifier.width(16.dp))
                        Text(
                            text = "Delete message",
                            color = Color(0xFFEF4444),
                            fontSize = 15.sp,
                            fontWeight = FontWeight.Medium
                        )
                    }
                }

                Spacer(modifier = Modifier.height(16.dp))
            }
        }
    }

    // Forward To... Bottom Sheet
    if (messageToForward != null) {
        val targetMsg = messageToForward!!
        val filteredChats = remember(allChats, forwardSearchQuery) {
            if (forwardSearchQuery.isBlank()) {
                allChats
            } else {
                allChats.filter { it.title.contains(forwardSearchQuery, ignoreCase = true) }
            }
        }

        ModalBottomSheet(
            onDismissRequest = {
                messageToForward = null
                forwardSearchQuery = ""
            },
            containerColor = GamEmpiresSurface
        ) {
            Column(
                modifier = Modifier
                    .fillMaxWidth()
                    .padding(horizontal = 20.dp, vertical = 12.dp)
            ) {
                Text(
                    text = "Forward To...",
                    fontSize = 18.sp,
                    fontWeight = FontWeight.Bold,
                    color = androidx.compose.material3.MaterialTheme.colorScheme.primary,
                    modifier = Modifier.padding(bottom = 12.dp)
                )

                OutlinedTextField(
                    value = forwardSearchQuery,
                    onValueChange = { forwardSearchQuery = it },
                    placeholder = {
                        Text("Search chats...", color = GamEmpiresTextMuted, fontSize = 14.sp)
                    },
                    singleLine = true,
                    shape = RoundedCornerShape(12.dp),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedContainerColor = GamEmpiresBg,
                        unfocusedContainerColor = GamEmpiresBg,
                        focusedBorderColor = GamEmpiresGold,
                        unfocusedBorderColor = GamEmpiresBorder,
                        focusedTextColor = MaterialTheme.colorScheme.onSurface,
                        unfocusedTextColor = MaterialTheme.colorScheme.onSurface
                    ),
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(bottom = 12.dp)
                )

                if (filteredChats.isEmpty()) {
                    Box(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(vertical = 24.dp),
                        contentAlignment = Alignment.Center
                    ) {
                        Text(
                            text = "No chats found",
                            color = GamEmpiresTextMuted,
                            fontSize = 14.sp
                        )
                    }
                } else {
                    LazyColumn(
                        modifier = Modifier
                            .fillMaxWidth()
                            .heightIn(max = 360.dp),
                        verticalArrangement = Arrangement.spacedBy(4.dp)
                    ) {
                        items(filteredChats, key = { it.id }) { chat ->
                            Row(
                                verticalAlignment = Alignment.CenterVertically,
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .clip(RoundedCornerShape(10.dp))
                                    .clickable {
                                        onForwardMessage(chat, targetMsg)
                                        messageToForward = null
                                        forwardSearchQuery = ""
                                    }
                                    .padding(vertical = 10.dp, horizontal = 8.dp)
                            ) {
                                Surface(
                                    shape = CircleShape,
                                    color = GamEmpiresGoldDark,
                                    modifier = Modifier.size(40.dp)
                                ) {
                                    Box(contentAlignment = Alignment.Center) {
                                        Text(
                                            text = chat.title.take(2).uppercase(),
                                            fontWeight = FontWeight.Bold,
                                            color = MaterialTheme.colorScheme.onSurface,
                                            fontSize = 14.sp
                                        )
                                    }
                                }
                                Spacer(modifier = Modifier.width(12.dp))
                                Column(modifier = Modifier.weight(1f)) {
                                    Text(
                                        text = chat.title,
                                        fontSize = 15.sp,
                                        fontWeight = FontWeight.SemiBold,
                                        color = GamEmpiresTextPrimary,
                                        maxLines = 1,
                                        overflow = TextOverflow.Ellipsis
                                    )
                                    val subtitle = if (chat.type == ChatType.DIRECT) "Direct chat" else "${chat.participantIds.size} members"
                                    Text(
                                        text = subtitle,
                                        fontSize = 12.sp,
                                        color = GamEmpiresTextMuted
                                    )
                                }
                            }
                        }
                    }
                }
                Spacer(modifier = Modifier.height(20.dp))
            }
        }
    }

    // Edit Message Dialog
    if (messageToEdit != null) {
        val target = messageToEdit!!
        var editedContent by rememberSaveable { mutableStateOf(target.content) }

        AlertDialog(
            onDismissRequest = { messageToEdit = null },
            title = {
                Text(text = "Edit Message", fontWeight = FontWeight.Bold)
            },
            text = {
                OutlinedTextField(
                    value = editedContent,
                    onValueChange = { editedContent = it },
                    singleLine = false,
                    maxLines = 5,
                    shape = RoundedCornerShape(12.dp),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedContainerColor = GamEmpiresBg,
                        unfocusedContainerColor = GamEmpiresBg,
                        focusedBorderColor = GamEmpiresGold,
                        unfocusedBorderColor = GamEmpiresBorder,
                        focusedTextColor = MaterialTheme.colorScheme.onSurface,
                        unfocusedTextColor = MaterialTheme.colorScheme.onSurface
                    ),
                    modifier = Modifier.fillMaxWidth()
                )
            },
            confirmButton = {
                TextButton(
                    onClick = {
                        val trimmed = editedContent.trim()
                        if (trimmed.isNotBlank()) {
                            onEditMessage(target.id, trimmed)
                        }
                        messageToEdit = null
                    }
                ) {
                    Text("Save", fontWeight = FontWeight.Bold, color = GamEmpiresGold)
                }
            },
            dismissButton = {
                TextButton(onClick = { messageToEdit = null }) {
                    Text("Cancel", color = GamEmpiresTextMuted)
                }
            }
        )
    }

    // Delete Message Confirmation Dialog
    if (messageToDelete != null) {
        val target = messageToDelete!!
        AlertDialog(
            onDismissRequest = { messageToDelete = null },
            title = {
                Text(text = "Delete Message?", fontWeight = FontWeight.Bold)
            },
            text = {
                Text(text = "Are you sure you want to delete this message? This cannot be undone.")
            },
            confirmButton = {
                TextButton(
                    onClick = {
                        onDeleteMessage(target.id)
                        messageToDelete = null
                    }
                ) {
                    Text("Delete", fontWeight = FontWeight.Bold, color = Color(0xFFEF4444))
                }
            },
            dismissButton = {
                TextButton(onClick = { messageToDelete = null }) {
                    Text("Cancel", color = GamEmpiresTextMuted)
                }
            }
        )
    }
}

/**
 * Requirement 4: Other system messages show as a small centered grey line, not a bubble.
 */
@Composable
private fun SystemMessageLine(
    content: String,
    modifier: Modifier = Modifier
) {
    Box(
        modifier = modifier
            .fillMaxWidth()
            .padding(vertical = 4.dp, horizontal = 24.dp),
        contentAlignment = Alignment.Center
    ) {
        Text(
            text = content,
            fontSize = 12.sp,
            color = GamEmpiresTextMuted,
            textAlign = TextAlign.Center
        )
    }
}

@OptIn(ExperimentalFoundationApi::class)
@Composable
private fun MessageBubble(
    message: Message,
    quotedMessage: Message?,
    isMe: Boolean,
    downloadProgress: Map<String, Float>,
    downloadedAttachmentIds: Set<String>,
    onLongClick: () -> Unit,
    onAddReaction: (String) -> Unit,
    onImageClick: (Attachment, String) -> Unit,
    onDownloadClick: (Attachment) -> Unit,
    onSaveToDownloads: (Attachment) -> Unit = onDownloadClick,
    modifier: Modifier = Modifier
) {
    // Requirement 4: A deleted message shows as one muted italic line
    if (message.isDeleted) {
        Row(
            modifier = modifier
                .fillMaxWidth()
                .padding(vertical = 4.dp, horizontal = 12.dp),
            horizontalArrangement = if (isMe) Arrangement.End else Arrangement.Start
        ) {
            Text(
                text = "This message was deleted",
                fontSize = 13.sp,
                fontStyle = FontStyle.Italic,
                color = GamEmpiresTextMuted
            )
        }
        return
    }

    val timeFormat = remember { SimpleDateFormat("h:mm a", Locale.getDefault()) }
    val formattedTime = remember(message.timestamp) { timeFormat.format(Date(message.timestamp)) }

    // Gaming look (1.0.11): angled bubbles. Mine in an amber frame, others with an amber edge on the left.
    val geDark = com.gamempires.messenger.ui.components.geIsDark()
    val bubbleBrush = androidx.compose.ui.graphics.Brush.linearGradient(
        when {
            isMe && geDark -> listOf(Color(0xFF2C220A), Color(0xFF1A150A))
            isMe -> listOf(Color(0xFFFFF3D6), Color(0xFFFFE7AE))
            geDark -> listOf(Color(0xFF1A1A1A), Color(0xFF131313))
            else -> listOf(Color.White, Color(0xFFFBF7EE))
        }
    )
    val bubbleShape = com.gamempires.messenger.ui.components.geBubbleShape(mine = isMe)
    val bubbleBorder = if (isMe) BorderStroke(1.5.dp, Color(0xFFFFAB00)) else BorderStroke(1.dp, if (geDark) Color(0x33FFAB00) else Color(0x40CC8800))
    val bodyTextColor = if (geDark) Color(0xFFEDEDED) else Color(0xFF1A1712)
    val metaColor = if (isMe) (if (geDark) Color(0xFFD9C9A0) else Color(0xFF7A5A10)) else MaterialTheme.colorScheme.onSurfaceVariant
    val tickColor = metaColor

    Column(
        horizontalAlignment = if (isMe) Alignment.End else Alignment.Start,
        modifier = modifier.fillMaxWidth()
    ) {
        Surface(
            shape = bubbleShape,
            color = Color.Transparent,
            border = bubbleBorder,
            modifier = Modifier
                .widthIn(max = 600.dp)
                .padding(
                    start = if (isMe) 48.dp else 0.dp,
                    end = if (isMe) 0.dp else 48.dp
                )
                .background(bubbleBrush, bubbleShape)
                .then(
                    if (isMe) Modifier else Modifier.drawBehind {
                        drawRect(Color(0xFFFFAB00), size = androidx.compose.ui.geometry.Size(3.dp.toPx(), size.height))
                    }
                )
                .combinedClickable(
                    onClick = {},
                    onLongClick = onLongClick
                )
        ) {
            Column(
                modifier = Modifier.padding(horizontal = 14.dp, vertical = 10.dp)
            ) {
                // Quoted reply banner if this message replied to another message
                if (quotedMessage != null) {
                    Surface(
                        shape = RoundedCornerShape(8.dp),
                        color = if (isMe) Color(0x22000000) else Color(0x33000000),
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(bottom = 6.dp)
                    ) {
                        Row(modifier = Modifier.padding(8.dp)) {
                            Box(
                                modifier = Modifier
                                    .width(3.dp)
                                    .height(30.dp)
                                    .background(Color(0xFFFFAB00))
                            )
                            Spacer(modifier = Modifier.width(8.dp))
                            Text(
                                text = quotedMessage.content,
                                fontSize = 12.sp,
                                maxLines = 2,
                                overflow = TextOverflow.Ellipsis,
                                color = bodyTextColor.copy(alpha = 0.8f)
                            )
                        }
                    }
                }

                // Requirement 3: Attachments (images inline with full screen tap, non-images as cards with download)
                if (message.attachments.isNotEmpty()) {
                    message.attachments.forEach { att ->
                        if (att.type == AttachmentType.IMAGE) {
                            val imgUrl = att.thumbnailUrl.takeIf { !it.isNullOrBlank() } ?: att.url
                            AsyncImage(
                                model = imgUrl,
                                contentDescription = att.name,
                                contentScale = ContentScale.Crop,
                                modifier = Modifier
                                    .fillMaxWidth()
                                    .height(180.dp)
                                    .clip(RoundedCornerShape(8.dp))
                                    .clickable { onImageClick(att, imgUrl) }
                            )
                            Spacer(modifier = Modifier.height(6.dp))
                        } else {
                            val progress = downloadProgress[att.id]
                            FileAttachmentCard(
                                attachment = att,
                                downloadProgress = progress,
                                downloaded = att.id in downloadedAttachmentIds,
                                onClick = { onDownloadClick(att) },
                                onSaveClick = { onSaveToDownloads(att) }
                            )
                            Spacer(modifier = Modifier.height(6.dp))
                        }
                    }
                }

                // Message Text: don't show duplicate file name or raw URL if attachment matches content
                val hasMatchingAttachment = message.attachments.any { att ->
                    val c = message.content.trim()
                    c.equals(att.name.trim(), ignoreCase = true) ||
                    c.equals(att.url.trim(), ignoreCase = true) ||
                    c == "{file}" ||
                    c.endsWith(att.name.trim(), ignoreCase = true)
                }
                if (!hasMatchingAttachment && message.content.isNotBlank()) {
                    MarkdownText(
                        markdown = message.content,
                        color = bodyTextColor,
                        linkColor = if (geDark) Color(0xFFFFC14D) else Color(0xFFA86A00),
                        fontSize = 14.sp
                    )
                }

                Spacer(modifier = Modifier.height(4.dp))

                // Time, Edited Status & Delivery Status
                Row(
                    verticalAlignment = Alignment.CenterVertically,
                    horizontalArrangement = Arrangement.End,
                    modifier = Modifier.align(Alignment.End)
                ) {
                    if (message.isEdited) {
                        Text(
                            text = "edited",
                            fontSize = 10.sp,
                            color = metaColor,
                            fontStyle = FontStyle.Italic

private fun formatFileSize(bytes: Long): String {
    if (bytes <= 0) return ""
    val kb = bytes / 1024.0
    val mb = kb / 1024.0
    val gb = mb / 1024.0
    return when {
        gb >= 1.0 -> String.format(Locale.US, "%.1f GB", gb)
        mb >= 1.0 -> String.format(Locale.US, "%.1f MB", mb)
        kb >= 1.0 -> String.format(Locale.US, "%.1f KB", kb)
        else -> "$bytes B"
    }
}

@Composable
private fun PendingFilesStrip(files: List<Uri>, onRemove: (Uri) -> Unit) {
    val context = LocalContext.current
    Column(
        modifier = Modifier
            .fillMaxWidth()
            .background(GamEmpiresBg)
            .padding(horizontal = 12.dp, vertical = 6.dp)
    ) {
        Text(
            text = if (files.size == 1) "1 file ready to send" else "${files.size} files ready to send",
            fontSize = 11.sp,
            fontWeight = FontWeight.Bold,
            color = GamEmpiresGold
        )
        Spacer(modifier = Modifier.height(4.dp))
        androidx.compose.foundation.lazy.LazyRow(
            horizontalArrangement = androidx.compose.foundation.layout.Arrangement.spacedBy(6.dp)
        ) {
            items(files.size, key = { files[it].toString() }) { index ->
                val uri = files[index]
                val name = remember(uri) {
                    runCatching {
                        context.contentResolver.query(uri, arrayOf(android.provider.OpenableColumns.DISPLAY_NAME), null, null, null)
                            ?.use { c -> if (c.moveToFirst()) c.getString(0) else null }
                    }.getOrNull() ?: uri.lastPathSegment ?: "file"
                }
                Surface(
                    shape = RoundedCornerShape(16.dp),
                    color = GamEmpiresCard,
                    border = BorderStroke(1.dp, GamEmpiresBorder)
                ) {
                    Row(
                        verticalAlignment = Alignment.CenterVertically,
                        modifier = Modifier.padding(start = 10.dp, end = 2.dp)
                    ) {
                        Icon(
                            imageVector = Icons.Default.AttachFile,
                            contentDescription = null,
                            tint = GamEmpiresGold,
                            modifier = Modifier.size(14.dp)
                        )
                        Spacer(modifier = Modifier.width(4.dp))
                        Text(
                            text = name,
                            fontSize = 12.sp,
                            color = MaterialTheme.colorScheme.onSurface,
                            maxLines = 1,
                            overflow = TextOverflow.Ellipsis,
                            modifier = Modifier.widthIn(max = 150.dp)
                        )
                        IconButton(onClick = { onRemove(uri) }, modifier = Modifier.size(30.dp)) {
                            Icon(
                                imageVector = Icons.Default.Close,
                                contentDescription = "Remove $name",
                                tint = GamEmpiresTextSecondary,

    fun downloadAndOpenFile(attachment: Attachment, context: Context) {
        if (!beginDownload(attachment.id)) return
        val downloadAccount = downloadedFiles.accountKey()
        viewModelScope.launch {

            try {
                downloadedFiles.existingUri(attachment.id, downloadAccount)?.let {
                    openContentUri(it, attachment.mimeType, context)
                    return@launch
                }
                val downloadsDir = File(context.cacheDir, "downloads").apply { mkdirs() }
                val targetFile = File(downloadsDir, attachment.name)
                val result = chatRepository.downloadFile(attachment, targetFile) { progress ->
                    _downloadProgress.update { it + (attachment.id to progress) }
                }

                if (result.isSuccess) {
                    saveFileToPublicDownloads(context, targetFile, attachment.name, attachment.mimeType)?.let {
                        downloadedFiles.markSaved(attachment.id, it, downloadAccount)

                    openFileWithMatchingApp(targetFile, attachment.mimeType, context)
                } else {
                    val err = result.exceptionOrNull()?.message ?: "Download failed"
                    Toast.makeText(context, "Failed to download ${attachment.name}: $err", Toast.LENGTH_LONG).show()
                }
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {

                Toast.makeText(context, "Failed to download ${attachment.name}: ${e.message}", Toast.LENGTH_LONG).show()
            } finally {
                _downloadProgress.update { it - attachment.id }
            }
        }
    }

    private suspend fun saveFileToPublicDownloads(
        context: Context,
        sourceFile: File,
        displayName: String,
        mimeType: String
    ): Uri? = withContext(Dispatchers.IO) {
        try {
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                val contentValues = ContentValues().apply {
                    put(MediaStore.MediaColumns.DISPLAY_NAME, displayName)
                    put(MediaStore.MediaColumns.MIME_TYPE, mimeType.ifBlank { "application/octet-stream" })
                    put(MediaStore.MediaColumns.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS + "/GamEmpires")
                    put(MediaStore.MediaColumns.IS_PENDING, 1)
                }
                val resolver = context.contentResolver
                val uri = resolver.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, contentValues) ?: return@withContext null
                try {
                    val stream = resolver.openOutputStream(uri) ?: throw java.io.IOException("Cannot write to Downloads")
                    stream.use { out ->
                        sourceFile.inputStream().use { input -> input.copyTo(out) }
                    }
                    contentValues.clear()
                    contentValues.put(MediaStore.MediaColumns.IS_PENDING, 0)
                    resolver.update(uri, contentValues, null, null)
                    uri
                } catch (e: Exception) {
                    resolver.delete(uri, null, null)
                    throw e
                }
            } else {
                val targetDir = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOWNLOADS)
                val gamEmpiresDir = File(targetDir, "GamEmpires").apply { mkdirs() }
                val targetFile = File(gamEmpiresDir, displayName)
                sourceFile.copyTo(targetFile, overwrite = true)
                Uri.fromFile(targetFile)
            }
        } catch (e: CancellationException) {
            throw e
        } catch (e: Exception) {
            android.util.Log.w("ChatDetailViewModel", "Could not copy to public downloads: ${e.message}")
            null
        }
    }

    private fun openFileWithMatchingApp(file: File, mimeType: String, context: Context) {
        try {
            val uri = FileProvider.getUriForFile(
                context,
                "${context.packageName}.fileprovider",
                file
            )
            openContentUri(uri, mimeType, context)
        } catch (e: Exception) {
            Toast.makeText(context, "Cannot open file: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    private fun openContentUri(uri: Uri, mimeType: String, context: Context) {
        try {
            val intent = Intent(Intent.ACTION_VIEW).apply {
                setDataAndType(uri, if (mimeType.isNotBlank()) mimeType else "*/*")
                addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION)
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            }
            context.startActivity(Intent.createChooser(intent, "Open with..."))
        } catch (e: ActivityNotFoundException) {
            Toast.makeText(context, "No app found to open this file", Toast.LENGTH_LONG).show()
        } catch (e: Exception) {
            Toast.makeText(context, "Cannot open file: ${e.message}", Toast.LENGTH_LONG).show()
        }
    }

    /** Sends the picked files one after another, so they arrive in the chat in the order picked. */
    fun uploadFiles(fileUris: List<Uri>, context: Context) {
        val id = _chatId.value ?: return
        viewModelScope.launch {
            fileUris.forEach { uploadOne(id, it, context) }
        }
    }

    fun uploadFile(fileUri: Uri, context: Context) = uploadFiles(listOf(fileUri), context)

    private suspend fun uploadOne(id: String, fileUri: Uri, context: Context) {
        run {
            try {
                var fileName = "file_${System.currentTimeMillis()}"
                var fileSize = 0L
                val mimeType = context.contentResolver.getType(fileUri) ?: "application/octet-stream"

                context.contentResolver.query(fileUri, null, null, null, null)?.use { cursor ->
                    val nameIndex = cursor.getColumnIndex(android.provider.OpenableColumns.DISPLAY_NAME)
                    val sizeIndex = cursor.getColumnIndex(android.provider.OpenableColumns.SIZE)
                    if (cursor.moveToFirst()) {
                        if (nameIndex != -1) {
                            val displayName = cursor.getString(nameIndex)
                            if (!displayName.isNullOrBlank()) fileName = displayName
                        }
                        if (sizeIndex != -1) {
                            fileSize = cursor.getLong(sizeIndex)
                        }
                    }
                }

                // The photo picker sometimes names files without an extension ("1000063123"); Nextcloud then
                // treats a photo as a plain file. Add the extension that matches the file type.
                if (!fileName.substringAfterLast('/').contains('.')) {
                    android.webkit.MimeTypeMap.getSingleton().getExtensionFromMimeType(mimeType)
                        ?.let { fileName = "$fileName.$it" }
                }

                _uploadProgress.value = _uploadProgress.value + (fileName to 0f)
                val result = chatRepository.uploadAndShareFile(
                    chatId = id,
                    fileUri = fileUri,
                    fileName = fileName,
                    mimeType = mimeType,
                    fileSize = fileSize,
                    context = context
                ) { progress ->
                    _uploadProgress.value = _uploadProgress.value + (fileName to progress)
                }

                _uploadProgress.value = _uploadProgress.value - fileName
                if (result.isFailure) {
                    val err = result.exceptionOrNull()?.message ?: "Upload failed"
                    Toast.makeText(context, "Upload failed: $err", Toast.LENGTH_LONG).show()
                }
            } catch (e: Exception) {
                Toast.makeText(context, "Cannot upload file: ${e.message}", Toast.LENGTH_LONG).show()
            }
        }
    }

    override fun onCleared() {
        super.onCleared()
        stopActiveSync()
    }
}

import android.app.Activity

/**
 * Optional fingerprint lock: when turned on in Settings, the app asks for the fingerprint (or the phone's
 * screen lock as a fallback) when it is opened, and again after it has been in the background for 30 s.
 * Incoming calls are never blocked by the lock.
 */
object AppLock {
    private const val PREFS = "gamempires_security"
    private const val KEY_ENABLED = "fingerprint_lock"
    private const val RELOCK_AFTER_MS = 30_000L

    private val _locked = MutableStateFlow(false)
    val locked: StateFlow<Boolean> = _locked.asStateFlow()
    private var backgroundSince = 0L // 0 = fresh process

    private fun prefs(context: Context) = context.applicationContext.getSharedPreferences(PREFS, Context.MODE_PRIVATE)

    fun isEnabled(context: Context) = prefs(context).getBoolean(KEY_ENABLED, false)

    fun setEnabled(context: Context, enabled: Boolean) {
        prefs(context).edit().putBoolean(KEY_ENABLED, enabled).apply()
        if (!enabled) _locked.value = false
    }

    /** True when the phone has a fingerprint/face or at least a screen lock set up. */
    fun isAvailable(context: Context): Boolean {
        return if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
            val manager = context.getSystemService(BiometricManager::class.java) ?: return false
            manager.canAuthenticate(AUTHENTICATORS) == BiometricManager.BIOMETRIC_SUCCESS
        } else {
            context.getSystemService(KeyguardManager::class.java)?.isDeviceSecure == true
        }
    }

    fun onForeground(context: Context, callLaunch: Boolean) {
        if (!isEnabled(context) || callLaunch) return
        val away = backgroundSince == 0L || SystemClock.elapsedRealtime() - backgroundSince > RELOCK_AFTER_MS
        if (away) _locked.value = true
    }

    fun onBackground() {
        backgroundSince = SystemClock.elapsedRealtime()
    }

    fun authenticate(context: Context, title: String, onSuccess: () -> Unit, onError: (String) -> Unit = {}) {
        val activity = context.findActivity() ?: return onError("Cannot show the fingerprint prompt here")
        val builder = BiometricPrompt.Builder(activity)
            .setTitle(title)
            .setSubtitle("Use your fingerprint or the phone's screen lock")
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
            builder.setAllowedAuthenticators(AUTHENTICATORS)
        } else {
            @Suppress("DEPRECATION")
            builder.setDeviceCredentialAllowed(true)
        }
        runCatching {
            builder.build().authenticate(CancellationSignal(), activity.mainExecutor,
                object : BiometricPrompt.AuthenticationCallback() {
                    override fun onAuthenticationSucceeded(result: BiometricPrompt.AuthenticationResult) = onSuccess()
                    override fun onAuthenticationError(errorCode: Int, errString: CharSequence) = onError(errString.toString())
                })
        }.onFailure { onError(it.message ?: "Fingerprint prompt failed") }
    }

    fun unlock(context: Context) {
        authenticate(context, "Unlock GamEmpires Messenger", onSuccess = { _locked.value = false })
    }

    private val AUTHENTICATORS: Int
        get() = BiometricManager.Authenticators.BIOMETRIC_WEAK or BiometricManager.Authenticators.DEVICE_CREDENTIAL

    private fun Context.findActivity(): Activity? {
        var c: Context? = this
        while (c is ContextWrapper) {
            if (c is Activity) return c
            c = c.baseContext
        }
        return null
    }
}

/** Covers the app until the fingerprint (or screen lock) is confirmed. */
@Composable
fun LockScreen() {
    val context = LocalContext.current
    LaunchedEffect(Unit) { AppLock.unlock(context) }
    Column(
        modifier = Modifier
            .fillMaxSize()
            .background(Color(0xFF141414))
            // Swallow touches so nothing underneath can be used while locked.
            .clickable(interactionSource = remember { MutableInteractionSource() }, indication = null) {},
        verticalArrangement = Arrangement.Center,
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Image(
            painter = painterResource(R.mipmap.ic_launcher_foreground),
            contentDescription = null,
            modifier = Modifier.size(200.dp)
        )
        Text("GamEmpires Messenger is locked", color = Color.White, fontSize = 18.sp)
        Spacer(Modifier.height(24.dp))
        Button(
            onClick = { AppLock.unlock(context) },
            colors = ButtonDefaults.buttonColors(containerColor = GamEmpiresGold, contentColor = Color.Black)
        ) {
            Icon(Icons.Default.Fingerprint, contentDescription = null)
            Spacer(Modifier.size(8.dp))
            Text("Unlock")
        }
    }
}

@Composable
fun AppearanceSettingsButton() {

    val setMode = LocalSetAppearance.current
    val context = LocalContext.current
    val scope = rememberCoroutineScope()
    var exporting by remember { mutableStateOf(false) }
    var showLicenses by remember { mutableStateOf(false) }

        Icon(Icons.Default.Settings, contentDescription = "Settings", tint = MaterialTheme.colorScheme.onSurface)
    }
    if (visible) AlertDialog(
        onDismissRequest = { visible = false },
        title = { Text("Settings") },
        text = {
            androidx.compose.foundation.layout.Column(
                Modifier.verticalScroll(androidx.compose.foundation.rememberScrollState())
            ) {
                SettingsSectionLabel("Appearance")
                AppearanceMode.entries.forEach { option ->
                    Row(
                        Modifier
                            .fillMaxWidth()
                            .selectable(selected = mode == option, role = Role.RadioButton, onClick = { setMode(option) })
                            .padding(vertical = 2.dp),
                        verticalAlignment = androidx.compose.ui.Alignment.CenterVertically
                    ) {
                        RadioButton(selected = mode == option, onClick = null)
                        Text(
                            when (option) {
                                AppearanceMode.SYSTEM -> "Use device setting"
                                AppearanceMode.LIGHT -> "Light"
                                AppearanceMode.DARK -> "Dark"

                    val intent = android.content.Intent(android.provider.Settings.ACTION_APP_NOTIFICATION_SETTINGS)
                        .putExtra(android.provider.Settings.EXTRA_APP_PACKAGE, context.packageName)
                    runCatching { context.startActivity(intent) }
                }
                // Android 14+: incoming calls can only take over the lock screen when this is allowed.
                if (android.os.Build.VERSION.SDK_INT >= 34) {
                    val notificationManager = context.getSystemService(android.app.NotificationManager::class.java)
                    if (notificationManager?.canUseFullScreenIntent() == false) {
                        SettingsRow("Allow call screen on lock screen") {
                            val intent = android.content.Intent(
                                android.provider.Settings.ACTION_MANAGE_APP_USE_FULL_SCREEN_INTENT,
                                android.net.Uri.parse("package:${context.packageName}")
                            )
                            runCatching { context.startActivity(intent) }
                        }
                    }
                }
                // Some phones (e.g. Xiaomi) also need "Show on lock screen" in the app's own permission page.
                SettingsRow("App permissions (lock screen, battery)") {
                    val intent = android.content.Intent(android.provider.Settings.ACTION_APPLICATION_DETAILS_SETTINGS,
                        android.net.Uri.parse("package:${context.packageName}"))
                    runCatching { context.startActivity(intent) }
                }

                HorizontalDivider(Modifier.padding(vertical = 12.dp))
                SettingsSectionLabel("Security")
                var lockOn by remember { mutableStateOf(com.gamempires.messenger.ui.security.AppLock.isEnabled(context)) }

                    Modifier
                        .fillMaxWidth()

                            color = MaterialTheme.colorScheme.onSurfaceVariant

                    }
                    Switch(checked = lockOn, onCheckedChange = { turnOn ->
                        val lock = com.gamempires.messenger.ui.security.AppLock
                        if (!turnOn) {
                            lock.authenticate(context, "Turn off fingerprint lock", onSuccess = {
                                lock.setEnabled(context, false); lockOn = false
                            })
                        } else if (!lock.isAvailable(context)) {
                            android.widget.Toast.makeText(context,
                                "Set up a fingerprint or screen lock in the phone settings first",
                                android.widget.Toast.LENGTH_LONG).show()
                        } else {
                            lock.authenticate(context, "Turn on fingerprint lock", onSuccess = {
                                lock.setEnabled(context, true); lockOn = true
                            })
                        }
                    })
                }

                HorizontalDivider(Modifier.padding(vertical = 12.dp))
                SettingsSectionLabel("About")
                SettingsRow(if (exporting) "Preparing diagnostics…" else "Share call diagnostics", enabled = !exporting) {
                    exporting = true
                    scope.launch {
                        try {
                            val file = com.gamempires.messenger.calls.CallDiagnostics.export(context)
                            val uri = androidx.core.content.FileProvider.getUriForFile(context, "${context.packageName}.fileprovider", file)
                            val intent = android.content.Intent(android.content.Intent.ACTION_SEND).apply {
                                type = "text/plain"
                                putExtra(android.content.Intent.EXTRA_STREAM, uri)
                                addFlags(android.content.Intent.FLAG_GRANT_READ_URI_PERMISSION)
                            }
                            context.startActivity(android.content.Intent.createChooser(intent, "Share call diagnostics"))
                        } catch (e: Exception) {
                            android.widget.Toast.makeText(context, "Could not export call diagnostics", android.widget.Toast.LENGTH_LONG).show()
                        } finally { exporting = false }
                    }
                }
                SettingsRow("Open-source licences") { showLicenses = true }
                Text(
                    "Version ${com.gamempires.messenger.BuildConfig.VERSION_NAME}",
                    style = MaterialTheme.typography.bodySmall,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.padding(top = 12.dp)
                )
            }
        },
        confirmButton = { TextButton(onClick = { visible = false }) { Text("Done") } }
    )
    if (showLicenses) OpenSourceLicensesDialog(onDismiss = { showLicenses = false })
}

@Composable
private fun SettingsSectionLabel(text: String) {

/** Full-width, left-aligned row so every option lines up with the section labels. */
@Composable
private fun SettingsRow(text: String, enabled: Boolean = true, onClick: () -> Unit) {
    Text(
        text,
        style = MaterialTheme.typography.bodyLarge,
        color = if (enabled) MaterialTheme.colorScheme.onSurface else MaterialTheme.colorScheme.onSurfaceVariant,
        modifier = Modifier
            .fillMaxWidth()
            .clickable(enabled = enabled, onClick = onClick)
            .padding(vertical = 12.dp)
    )
}

@Composable
private fun OpenSourceLicensesDialog(onDismiss: () -> Unit) {
    val context = LocalContext.current
    val licenses = remember {
        runCatching {
            context.assets.open("open_source_licenses.txt").bufferedReader().use { it.readText() }
        }.getOrDefault("Licence information is not available.")
    }
    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text("Open-source licences") },
        text = {
            androidx.compose.foundation.layout.Box(
                Modifier.verticalScroll(androidx.compose.foundation.rememberScrollState())
            ) {
                Text(licenses, style = MaterialTheme.typography.bodySmall)
            }
        },
        confirmButton = { TextButton(onClick = onDismiss) { Text("Close") } }
    )
}

@Composable
fun MyApplicationTheme(content: @Composable () -> Unit) {
    val context = LocalContext.current
    val preferences = remember { context.getSharedPreferences("appearance", Context.MODE_PRIVATE) }
    var mode by remember { mutableStateOf(AppearanceMode.fromPreference(preferences.getString("mode", null))) }
    val dark = when (mode) {
        AppearanceMode.DARK -> true
        AppearanceMode.LIGHT -> false
        AppearanceMode.SYSTEM -> isSystemInDarkTheme()
    }
    CompositionLocalProvider(
        LocalAppearanceMode provides mode,
        LocalSetAppearance provides { selected ->
            mode = selected
            preferences.edit().putString("mode", selected.name).apply()
        }
    ) {
        MaterialTheme(colorScheme = if (dark) DarkAppColors else LightAppColors, typography = Typography, content = content)
    }
}
